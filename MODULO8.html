8.2 base.html — herança de template

<!DOCTYPE html>
<html lang="pt-BR">
<head>
  <meta charset="UTF-8">
  <meta name="viewport" content="width=device-width, initial-scale=1.0">
  <title>{% block titulo %}Plataforma Bling{% endblock %}</title>
  <link rel="stylesheet" href="{{ url_for('static', path='/css/style.css') }}">
</head>
<body>
  <nav>
    <a href="/dashboard">Dashboard</a>
    <a href="/pedidos/">Pedidos</a>
    <a href="/produtos/">Produtos</a>
    <a href="/nfs/historico">Histórico</a>
    <form action="/auth/logout" method="post" style="display:inline">
      <button type="submit">Sair</button>
    </form>
  </nav>

  {% if erro %}<div class="alerta alerta-erro">{{ erro }}</div>{% endif %}

  <main>{% block conteudo %}{% endblock %}</main>
</body>
</html>
Detalhe importante

Logout é POST, não link — operações que mudam estado nunca devem ser GET: o prefetch do navegador deslogaria o usuário.

url_for('static', path=...) gera a URL a partir do app.mount("/static", ...).

{% extends "base.html" %}

{% block conteudo %}
<h1>Pedidos prontos para faturar</h1>

{% if not pedidos %}
  <p>Nenhum pedido em aberto.</p>
{% else %}
<form action="/pedidos/emitir" method="post">
  <table>
    <thead>
      <tr><th></th><th>Nº</th><th>Data</th><th>Cliente</th><th>Total</th></tr>
    </thead>
    <tbody>
      {% for p in pedidos %}
      <tr>
        <!-- name repetido → FastAPI resolve como List[str] -->
        <td><input type="checkbox" name="pedido_ids" value="{{ p.id }}"></td>
        <td>{{ p.numero }}</td>
        <td>{{ p.data }}</td>
        <td>{{ p.contato.nome if p.contato else '—' }}</td>
        <td>R$ {{ '%.2f'|format(p.totalVenda or 0) }}</td>
      </tr>
      {% endfor %}
    </tbody>
  </table>

  <label>E-mail para envio dos documentos
    <input type="email" name="email_destino" required>
  </label>

  <label>Data de operação (opcional)
    <input type="date" name="data_base">
  </label>

  <button type="submit">Emitir NF-e das selecionadas</button>
</form>
{% endif %}
{% endblock %}
O detalhe do name repetido

O HTML envia pedido_ids=1&pedido_ids=2&pedido_ids=3. O FastAPI, vendo a anotação List[str] = Form(...), agrupa automaticamente. É por isso que python-multipart está nas dependências.

data_base vazio chega como "" — daí o validator data_base_vazia_vira_none no schema.

8.6 Segurança: XSS e o autoescape do Jinja2

Jinja2 escapa HTML por padrão em arquivos .html. Se um cliente se chamar <script>alert(1)</script>, será renderizado como texto.

Nunca faça

Não use |safe em dados vindos do Bling ou do usuário. Isso desliga o escape e abre XSS.

8.7 Proteção CSRF (lacuna atual)

Seus forms POST não têm token CSRF. Com samesite="lax" o risco é reduzido, mas a proteção completa exige:

# Gerar no GET
csrf = secrets.token_hex(16)
response.set_cookie("csrf_token", csrf, httponly=True, samesite="lax")

# Validar no POST
def validar_csrf(request: Request, csrf_token: str = Form(...)):
    if request.cookies.get("csrf_token") != csrf_token:
        raise HTTPException(403, "Token CSRF inválido")
<input type="hidden" name="csrf_token" value="{{ csrf_token }}">

8.8 Paginação (resolvendo o limite de 100)

async def get_todos_produtos(access_token: str, categoria_id: str) -> list[dict]:
    """Percorre todas as páginas até a Bling retornar lista vazia."""
    todos, pagina = [], 1
    async with httpx.AsyncClient(timeout=30.0) as client:
        while True:
            resp = await client.get(
                f"{BLING_BASE_URL}/produtos",
                headers=_bearer(access_token),
                params={"idCategoria": categoria_id, "pagina": pagina, "limite": 100},
            )
            if resp.is_error:
                raise ValueError(f"Erro ao listar produtos: {_extrair_erro_bling(resp)}")

            dados = resp.json().get("data", [])
            if not dados:
                break
            todos.extend(dados)
            pagina += 1
            await asyncio.sleep(0.4)      # rate limit
            if pagina > 50:               # guarda contra loop infinito
                break
    return todos

8.9 Emissão em background (resolvendo o timeout de 3 min)

Hoje 20 pedidos ≈ 3 minutos de requisição HTTP → risco de 504 em proxies.

from fastapi import BackgroundTasks
from db.database import AsyncSessionLocal

@router.post("/emitir")
async def emitir_nfs(
    request: Request,
    background_tasks: BackgroundTasks,
    pedido_ids: List[str] = Form(...),
    email_destino: str = Form(...),
    data_base: str = Form(None),
):
    usuario_id = get_usuario_id(request)
    if not usuario_id:
        return RedirectResponse("/auth/login")

    async def processar():
        # Sessão própria: a de Depends(get_db) já foi fechada
        async with AsyncSessionLocal() as db:
            await nf_service.emitir_nfs_massa(
                db=db, usuario_id=usuario_id, pedido_ids=pedido_ids,
                data_base=data_base, email_destino=email_destino,
            )

    background_tasks.add_task(processar)
    return RedirectResponse("/nfs/historico?processando=1", status_code=303)

O usuário acompanha o resultado em /nfs/historico, que já lê de operacoes_nf. Para produção com múltiplos workers, evolua para Celery ou ARQ com Redis.

8.10 Testes

# tests/test_nf_service.py
import pytest
from services.nf_service import _validar_pedido_basico, _montar_payload_nf

def test_pedido_sem_cpf():
    assert "sem CPF/CNPJ" in _validar_pedido_basico({"contato": {"nome": "X"}})

def test_pedido_sem_itens():
    pedido = {"contato": {"nome": "X", "numeroDocumento": "12345678901"}, "itens": []}
    assert "sem itens" in _validar_pedido_basico(pedido)

def test_pedido_valido():
    pedido = {
        "contato": {"nome": "X", "numeroDocumento": "123.456.789-01"},
        "itens": [{"descricao": "P", "valor": 10.0, "quantidade": 1}],
    }
    assert _validar_pedido_basico(pedido) is None

def test_payload_cria_parcela_quando_pedido_nao_tem():
    pedido = {
        "contato": {"nome": "X", "numeroDocumento": "12345678901"},
        "itens": [{"codigo": "A", "descricao": "P", "valor": 100.0, "quantidade": 1}],
        "parcelas": [], "total": 100.0,
    }
    payload = _montar_payload_nf(pedido, {}, "2026-08-11", "999")
    assert len(payload["parcelas"]) == 1
    assert payload["parcelas"][0]["valor"] == 100.0
    assert "Pedido Original: 999" in payload["observacoes"]
