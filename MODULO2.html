Banco de dados: SQLAlchemy 2.0 async + PostgreSQL

Engine assíncrono com asyncpg, models de auditoria, criação de tabelas e migrações com Alembic.

2.1 O engine assíncrono

# db/database.py
from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker
from sqlalchemy.orm import declarative_base
from dotenv import load_dotenv
import os

load_dotenv()

DATABASE_URL = os.getenv("DATABASE_URL")  # postgresql+asyncpg://...

engine = create_async_engine(
    DATABASE_URL,
    echo=False,          # True apenas para debug de SQL
    pool_pre_ping=True,  # testa a conexão antes de usar
    pool_size=10,
    max_overflow=20,
)

AsyncSessionLocal = async_sessionmaker(
    bind=engine,
    class_=AsyncSession,
    expire_on_commit=False,  # crítico em async
)

Base = declarative_base()

async def get_db():
    async with AsyncSessionLocal() as session:
        yield session

Anatomia da URL:

postgresql+asyncpg://user:senha@host:5432/banco └─dialeto─┘ └driver┘
Conceito-chave

Por que expire_on_commit=False? Por padrão o SQLAlchemy marca objetos como “expirados” após commit(). Ao acessar usuario.email depois, ele tentaria um novo SELECT — impossível de forma implícita em contexto async, gerando MissingGreenlet.

2.2 Models — versão PostgreSQL

Diferenças frente ao MySQL: Enum nativo precisa de name=; Numeric(10,2) retorna Decimal.

# db/models.py
import enum
from sqlalchemy import (Column, Integer, String, Text, Boolean,
                        DateTime, Enum, Numeric, ForeignKey, Index)
from sqlalchemy.orm import relationship
from sqlalchemy.sql import func
from db.database import Base


class StatusNF(str, enum.Enum):
    emitida = "emitida"
    erro = "erro"


class Usuario(Base):
    __tablename__ = "usuarios"

    id         = Column(Integer, primary_key=True, autoincrement=True)
    email      = Column(String(255), unique=True, nullable=False, index=True)
    senha_hash = Column(String(255), nullable=False)
    created_at = Column(DateTime(timezone=True), server_default=func.now())

    tokens             = relationship("BlingToken", back_populates="usuario",
                                      cascade="all, delete-orphan")
    operacoes_nf       = relationship("OperacaoNF", back_populates="usuario",
                                      cascade="all, delete-orphan")
    atualizacoes_preco = relationship("AtualizacaoPreco", back_populates="usuario",
                                      cascade="all, delete-orphan")


class BlingToken(Base):
    __tablename__ = "bling_tokens"

    id            = Column(Integer, primary_key=True, autoincrement=True)
    usuario_id    = Column(Integer, ForeignKey("usuarios.id", ondelete="CASCADE"),
                           nullable=False)
    access_token  = Column(Text, nullable=False)
    refresh_token = Column(Text, nullable=False)
    expires_at    = Column(DateTime(timezone=True), nullable=False)
    ativo         = Column(Boolean, default=True, nullable=False)

    usuario = relationship("Usuario", back_populates="tokens")

    # Acelera o SELECT de token ativo (usado em toda requisição à Bling)
    __table_args__ = (Index("ix_token_usuario_ativo", "usuario_id", "ativo"),)


class OperacaoNF(Base):
    __tablename__ = "operacoes_nf"

    id              = Column(Integer, primary_key=True, autoincrement=True)
    usuario_id      = Column(Integer, ForeignKey("usuarios.id", ondelete="CASCADE"),
                             nullable=False)
    pedido_id_bling = Column(String(50), nullable=False)
    nf_id_bling     = Column(String(50), nullable=True)   # None se falhou
    status          = Column(Enum(StatusNF, name="status_nf"), nullable=False)
    erro_descricao  = Column(Text, nullable=True)
    email_enviado   = Column(String(255), nullable=True)
    created_at      = Column(DateTime(timezone=True), server_default=func.now())

    usuario = relationship("Usuario", back_populates="operacoes_nf")

    __table_args__ = (Index("ix_nf_usuario_data", "usuario_id", "created_at"),)


class AtualizacaoPreco(Base):
    __tablename__ = "atualizacoes_preco"

    id               = Column(Integer, primary_key=True, autoincrement=True)
    usuario_id       = Column(Integer, ForeignKey("usuarios.id", ondelete="CASCADE"),
                              nullable=False)
    produto_id_bling = Column(String(50), nullable=False)
    nome_produto     = Column(String(255), nullable=True)
    preco_anterior   = Column(Numeric(10, 2), nullable=False)
    preco_novo       = Column(Numeric(10, 2), nullable=False)
    created_at       = Column(DateTime(timezone=True), server_default=func.now())

    usuario = relationship("Usuario", back_populates="atualizacoes_preco")

2.3 Modelo relacional

usuarios (1) ──┬──(N) bling_tokens [só 1 com ativo=true] ├──(N) operacoes_nf [auditoria de NF] └──(N) atualizacoes_preco [auditoria de preço]

As tabelas de operação são log de auditoria: nunca são atualizadas, só inseridas. É isso que permite o historico_nf (veja nfs.py) e os contadores do dashboard sem chamar a API Bling.

# routers/nfs.py — consulta direta ao banco, não bate na Bling
result = await db.execute(
    select(OperacaoNF)
    .where(OperacaoNF.usuario_id == usuario_id)
    .order_by(OperacaoNF.created_at.desc())   # mais recentes primeiro
)
operacoes = result.scalars().all()

2.4 Criação de tabelas no startup

@asynccontextmanager
async def lifespan(app: FastAPI):
    async with engine.begin() as conn:
        await conn.run_sync(Base.metadata.create_all)
    yield
    await engine.dispose()   # ← acrescente: fecha o pool no shutdown

run_sync é necessário porque create_all é uma API síncrona; ele a executa numa thread compatível com o event loop.

Limitação

create_all não altera tabelas existentes. Se você mudar uma coluna, ela não será modificada. Para produção → Alembic.
