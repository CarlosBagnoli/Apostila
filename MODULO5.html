O adaptador único de HTTP, timeouts proporcionais, tradução de erros da Bling e o desafio do DANFE em três camadas de codificação.

5.1 Princípio do adaptador único

Um só arquivo faz httpx. Benefícios: um lugar para mudar timeout, um lugar para tratar erro, e todo o resto do sistema fica testável com mock.

5.2 Padrão de todo método

async def get_pedido(access_token: str, pedido_id: str) -> dict:
    async with httpx.AsyncClient(timeout=15.0) as client:   # 1. fecha conexões
        try:
            response = await client.get(
                f"{BLING_BASE_URL}/pedidos/vendas/{pedido_id}",
                headers=_bearer(access_token),
            )
        except httpx.ReadTimeout:                            # 2. erro de rede
            raise ValueError(f"Timeout ao buscar pedido {pedido_id}.")

    if response.is_error:                                    # 3. erro HTTP
        raise ValueError(f"Pedido não encontrado: {_extrair_erro_bling(response)}")

    return response.json().get("data", {})                   # 4. desembrulha "data"

Sempre desembrulhe data. A Bling v3 responde {"data": {...}}. Usar .get("data", {}) evita KeyError quando a resposta é 204 No Content.

5.3 Tradução de erros

def _extrair_erro_bling(response) -> str:
    body = response.json()
    err = body.get("error", {})
    code = err.get("type") or err.get("code") or ""
    msg  = err.get("message") or err.get("description") or ""
    fields = err.get("fields") or []
    partes = [f"{f.get('name')}: {f.get('msg')}" for f in fields if isinstance(f, dict)]
    return " | ".join(x for x in ([code, msg] + partes) if x) or f"HTTP {response.status_code}"

Transforma {"error":{"type":"VALIDATION_ERROR","fields":[{"name":"itens","msg":"..."}]}} em texto legível que vai direto para operacoes_nf.erro_descricao e para a tela. Sem essa função o usuário veria apenas “HTTP 400”.

5.5 Estudo de caso: DANFE em três camadas

O download exige entender que o Bling encapsula o PDF em JSON → Base64URL → GZIP → bytes:

# 1. Buscar a chaveAcesso (44 dígitos) — o endpoint de documento usa a chave, não o id
info = await client.get(f"{BLING_BASE_URL}/nfe/{nf_id}", headers=_bearer(token))
chave_acesso = info.json().get("data", {}).get("chaveAcesso")

# 2. Baixar
response = await client.get(
    f"{BLING_BASE_URL}/nfe/documento/{chave_acesso}",
    params={"formato": "pdf"}, headers=_bearer(token),
)

# 3. Desembrulhar lista/dict
conteudo = response.json().get("data")
if isinstance(conteudo, list) and conteudo:
    item = conteudo[0]
    conteudo = (item.get("conteudo") or list(item.values())[0]) \
               if isinstance(item, dict) else item

# 4. Limpar quebras de linha
conteudo = conteudo.strip().replace("\n","").replace("\r","").replace(" ","")

# 5. Corrigir padding Base64 (múltiplo de 4)
resto = len(conteudo) % 4
if resto:
    conteudo += "=" * (4 - resto)

# 6. Base64URL (altchars=b'-_') → bytes comprimidos
dados = base64.b64decode(conteudo, altchars=b'-_')

# 7. GZIP
try:
    return gzip.decompress(dados)
except Exception:
    return dados     # fallback: não estava comprimido
PassoPor quê
PaddingBase64 exige comprimento múltiplo de 4; o Bling às vezes omite os =
altchars=b'-_'Variante URL-safe (- e _ em vez de + e /)
GZIPUm DANFE de ~200 KB viaja como ~40 KB
try/except no gzipContrato não garantido pela Bling

a) Imports mal posicionados

import base64 e import gzip estão indentados no meio do arquivo, após um return em enviar_nf — código morto que nunca executa. Mova para o topo:

import base64

b) get_danfe e get_xml são ~95% idênticos

async def _baixar_documento(access_token: str, nf_id: str, formato: str) -> bytes:
    """formato: 'pdf' (DANFE) ou 'xml'. Único ponto de decode Base64+GZIP."""
    async with httpx.AsyncClient(timeout=30.0) as client:
        info = await client.get(f"{BLING_BASE_URL}/nfe/{nf_id}",
                                headers=_bearer(access_token))
        if info.is_error:
            raise ValueError(f"Falha ao buscar dados da NF {nf_id}.")

        chave = info.json().get("data", {}).get("chaveAcesso")
        if not chave:
            raise ValueError(f"Chave de acesso ausente para a NF {nf_id} "
                             "(a NF pode não estar autorizada ainda).")

        resp = await client.get(
            f"{BLING_BASE_URL}/nfe/documento/{chave}",
            params={"formato": formato}, headers=_bearer(access_token),
        )

    if resp.is_error:
        raise ValueError(f"{formato.upper()} indisponível: {_extrair_erro_bling(resp)}")

    conteudo = resp.json().get("data")
    if isinstance(conteudo, list) and conteudo:
        item = conteudo[0]
        conteudo = (item.get("conteudo") or list(item.values())[0]) \
                   if isinstance(item, dict) else item

    if not isinstance(conteudo, str):
        raise ValueError(f"Formato inesperado do Bling: {str(conteudo)[:100]}")

    conteudo = conteudo.strip().replace("\n", "").replace("\r", "").replace(" ", "")
    if resto := len(conteudo) % 4:
        conteudo += "=" * (4 - resto)

    dados = base64.b64decode(conteudo, altchars=b"-_")
    try:
        return gzip.decompress(dados)
    except Exception:
        return dados


async def get_danfe(access_token: str, nf_id: str) -> bytes:
    return await _baixar_documento(access_token, nf_id, "pdf")

async def get_xml(access_token: str, nf_id: str) -> bytes:
    return await _baixar_documento(access_token, nf_id, "xml")
